\section{\textcolor{AccentGold}{سوال اول}}

جملات زیر را به فرم منطق مرتبه اول بنویسید:

\begin{enumerate}
	\item علی یا حداقل یک کتاب تاریخ و حداقل یک کتاب فلسفه دارد، یا غمگین است. (نه هر دو در یک زمان)
	\item فقط یک نفر به جز خودِ علی وجود دارد که هر چیزی را که علی دوست دارد، دوست داشته باشد و آن فرد سینا است.
	\item برخی از مردم در یک زمان مشخص شام می‌خورند، اما همه‌ی مردم در همه‌ی زمان‌ها شام نمی‌خورند.
	\item در هر مسابقه، حتماً یکی از دو بازیکن برنده می‌شوند.
	\item تمام درس‌های دانشگاه، سخت و بی‌فایده نیستند.
	\item هر ماشینی به جز پراید و پیکان، امن و با کیفیت است.
\end{enumerate}

\section{\textcolor{AccentGold}{سوال دوم}}
در یک پروندهٔ جنایی، با داده‌های به‌دست‌آمده از طرف یک کارآگاه می‌خواهیم مدلی طراحی کنیم که با استفاده از استنتاج زنجیره‌ای رو‌به‌عقب \lr{(Backward Chaining)} تعیین می‌کند که آیا \lr{John} مظنون جنایت است یا خیر. گام‌های استنتاج را نشان دهید.


\begin{latin}
\textbf{Facts}
\begin{itemize}
	\item Fact 1: Witness1(Saw, John, CrimeScene)
	\item Fact 2: Witness2(Saw, Jane, CrimeScene)
	\item Fact 3: Witness2(Saw, John, CrimeScene)
	\item Fact 4: Fingerprint(John, Gun), Motive(John, FinancialGain)
	\item Fact 5: Motive(Jane, Revenge)
	\item Fact 6: Weapon(Gun)
	\item Fact 7: Time(Crime, Night)
\end{itemize}

\textbf{Rules}
\begin{itemize}
	\item Rule A: If Witness1(Saw, X, CrimeScene) and Witness2(Saw, X, CrimeScene) 
	then PresentAtCrimeScene(X).
	\item Rule B: If Fingerprint(X, Gun) and Weapon(Gun), then UsedWeapon(X, Gun).
	\item Rule C: If Motive(X, FinancialGain) then HasMotive(X).
	\item Rule D: If Motive(X, Revenge) then HasMotive(X).
	\item Rule E: If PresentAtCrimeScene(X) and UsedWeapon(X, Gun) and HasMotive(X), then Suspect(X).
\end{itemize}
\end{latin}

\section{\textcolor{AccentGold}{سوال سوم}}
با استفاده از \lr{backward chaining} بررسی کنید که آیا \lr{Q} در پایگاه‌دانش زیر صحیح است یا خیر؟
\begin{latin}
\begin{itemize}
	\item ${P} \rightarrow {Q}$
	\item ${E} \rightarrow {B}$
	\item ${R} \rightarrow {Q}$
	\item ${M} \land {N} \rightarrow {Q}$
	\item ${A} \land {B} \rightarrow {P}$
	\item ${A} \rightarrow {M}$
	\item ${C} \rightarrow {M}$
	\item ${D} \rightarrow {N}$
	\item ${D}$
	\item ${A}$
\end{itemize}
\end{latin}

\section{\textcolor{AccentGold}{سوال پنجم}}
گزاره‌های زیر را به منطق مرتبه اول تبدیل کنید و سپس نتیجه را با کمک \lr{Resolution} استنتاج کنید.

\textbf{مسئله ۱}
\begin{enumerate}
	\item هر کودکی، سانتا را دوست دارد.
	\item هر کسی که سانتا را دوست دارد، هر گوزن شمالی‌ای را دوست دارد.
	\item \lr{Rudolph} یک گوزن شمالی است و \lr{Rudolph} بینی قرمز دارد.
	\item هر چیزی که بینی قرمز داشته باشد، یا عجیب است یا دلقک است.
	\item هیچ گوزن شمالی‌ای دلقک نیست.
	\item \lr{Scrooge} هیچ چیز عجیبی را دوست ندارد.
	\item (نتیجه) \lr{Scrooge} کودک نیست.
\end{enumerate}

\textbf{مسئله ۲}
\begin{enumerate}
	\item هر کسی که پنیر بخرد، یا یک موش دارد یا یک فروشگاه مواد غذایی دارد.
	\item هر گربه‌ای، یک موش را تعقیب می‌کند.
	\item \lr{Peter} پنیر می‌خرد.
	\item هر کسی که یک موش داشته باشد، از هر چیزی که هر موشی را تعقیب کند متنفر است.
	\item \lr{Alex} یک گربه دارد.
	\item کسی که از چیزی که متعلق به شخص دیگری است متنفر باشد، با آن شخص دوست نخواهد شد.
	\item (نتیجه) اگر \lr{Peter} یک فروشگاه مواد غذایی نداشته باشد، با \lr{Alex} دوست نخواهد شد.
\end{enumerate}

\section{\textcolor{AccentGold}{سوال ششم}}

در جزیره‌ی \lr{Knights and Knaves}، شوالیه‌ها همیشه حقیقت را می‌گویند و دروغگوها همیشه دروغ می‌گویند، اما در ابتدا مشخص نیست چه کسی شوالیه است و چه کسی دروغگو. شما با \lr{Albert}، \lr{Betty} و \lr{Carla} روبه‌رو می‌شوید که هر کدام یا شوالیه هستند یا دروغگو.

\begin{itemize}
	\item \lr{Albert} می‌گوید: «\lr{Betty} شوالیه است، اما \lr{Carla} دروغگو است.»
	\item \lr{Betty} می‌گوید: «دو نفر از ما شوالیه هستند.»
	\item \lr{Carla} می‌گوید: «من و \lr{Albert} هر دو شوالیه هستیم.»
\end{itemize}

ثابت کنید که \lr{Carla} یک دروغگو است. (اطمینان حاصل کنید که تمام مراحل استدلال را توضیح می‌دهید و قواعدی را که در هر مرحله استفاده کرده‌اید، به‌وضوح مشخص می‌کنید.)
