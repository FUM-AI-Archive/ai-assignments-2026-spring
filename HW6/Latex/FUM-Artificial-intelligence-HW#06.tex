\section{\textcolor{AccentGold}{سوال اول}}
\noindent هر یک از گزاره‌های زیر را اثبات کنید.

\begin{latin}
\begin{enumerate}
	\item $\alpha$ is satisfiable iff $\neg\alpha$ is not valid.
	\item $\alpha \models \gamma$ iff the sentence $(\alpha \rightarrow \gamma)$ is valid.
	\item $\alpha \equiv \neg\beta$ iff $(\alpha \leftrightarrow \neg\beta)$ is valid.
	\item $\alpha \models \beta$ iff $(\alpha \wedge \neg\beta)$ is unsatisfiable.
	\item $\alpha \equiv \beta$ iff $(\alpha \oplus \beta)$ is unsatisfiable.
	\item $(\alpha \vee \beta)$ is valid iff both $\neg\alpha$ and $\neg\beta$ are unsatisfiable.
	\item $(\alpha \wedge (\alpha \rightarrow \beta)) \models \beta$ iff $(\alpha \wedge \neg\beta)$ is unsatisfiable.
\end{enumerate}
\end{latin}

\section{\textcolor{AccentGold}{سوال دوم}}

\noindent الف. تمام عبارات زیر را به فرم CNF تبدیل کنید.

\begin{latin}
\begin{enumerate}
	\item $A \Leftrightarrow ((B \vee C) \Rightarrow D)$
	\item $D \Rightarrow (E \wedge \neg F)$
	\item $(A \wedge E) \Rightarrow G$
	\item $\neg G \vee (F \Rightarrow H)$
	\item $H \Leftrightarrow (C \vee \neg D)$
	\item $B \Rightarrow (\neg A \vee F)$
\end{enumerate}
\end{latin}
\noindent ب. با استفاده از روش Resolution، جمله زیر را از مجموعه‌ی جملات به‌دست‌آمده در بخش الف اثبات کنید یا رد کنید:

\begin{latin}
\begin{enumerate}
	\item $\neg A \wedge \neg H \Rightarrow \neg B$
\end{enumerate}
\end{latin}

\section{\textcolor{AccentGold}{سوال سوم}}

\noindent درستی یا نادرستی گزاره‌های زیر را اثبات کنید.

\begin{latin}
\begin{enumerate}
	\item $(A \wedge (A \rightarrow C) \wedge (B \vee \neg C \rightarrow D)) \models D$
	\item $(A \vee \neg B) \wedge (\neg A \vee D) \wedge (B \rightarrow E) \wedge (\neg E \vee G) \models G$
	\item $A \wedge (A \rightarrow B) \wedge ((B \rightarrow C) \rightarrow D) \models D$
	\item $\neg B \wedge (B \rightarrow C) \models \neg C$
\end{enumerate}
\end{latin}

\section{\textcolor{AccentGold}{سوال چهارم}}
\noindent عبارات زیر را به گزاره‌های منطق مرتبۀ اول تبدیل کنید.

\begin{latin}
\begin{enumerate}
	\item Every researcher who advises at least two students supervises all projects that involve one of their students.
	\begin{itemize}
		\item[] Predicates: $R(x)$: $x$ is a researcher
		\item[] $S(x,y)$: $x$ is a student of $y$
		\item[] $P(x,y)$: project $x$ involves $y$
		\item[] $V(x,y)$: $x$ supervises project $y$
	\end{itemize}
	
	\item There exists exactly one doctor who treats every patient that no nurse treats.
	\begin{itemize}
		\item[] Predicates: $D(x)$: $x$ is a doctor
		\item[] $N(x)$: $x$ is a nurse
		\item[] $T(x,y)$: $x$ treats $y$
	\end{itemize}
	
	\pagebreak
	
	\item Any course that is taught by both a senior professor and a junior professor has at least three students enrolled in it.
	\begin{itemize}
		\item[] Predicates: $C(x)$: $x$ is a course
		\item[] $S(x)$: $x$ is a student
		\item[] $Prof(x)$: $x$ is a professor
		\item[] $J(x)$: $x$ is junior
		\item[] $T(x,y)$: $x$ teaches $y$
		\item[] $E(x,y)$: $x$ is enrolled in $y$
	\end{itemize}
	
	\item No more than four animals in the shelter are unvaccinated, and every unvaccinated animal must be examined by all veterinarians.
	\begin{itemize}
		\item[] Predicates: $A(x)$: $x$ is an animal
		\item[] $U(x)$: $x$ is unvaccinated
		\item[] $V(x)$: $x$ is a veterinarian
		\item[] $E(x,y)$: $x$ examines $y$
	\end{itemize}
	
	\item There is a unique city in which every citizen owns at least one building, and any building in that city has exactly one owner.
	\begin{itemize}
		\item[] Predicates: $City(x)$: $x$ is a city
		\item[] $Person(x)$: $x$ is a person
		\item[] $B(x)$: $x$ is a building
		\item[] $Own(x,y)$: $x$ owns $y$
		\item[] $In(x,y)$: $y$ is in $x$
	\end{itemize}
\end{enumerate}
\end{latin}
