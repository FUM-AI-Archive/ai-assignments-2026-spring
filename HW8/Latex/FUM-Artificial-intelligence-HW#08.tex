	\section{\textcolor{AccentGold}{سؤال دوم}}
	
	فرض کنید یک پهپاد (\lr{Drone}) وظیفه دارد یک محموله امدادی (\lr{Cargo}) را از کمپ اصلی در موقعیت \lr{Base} به دست یک تیم نجات در موقعیت \lr{Camp} برساند. در ابتدا پهپاد و محموله هر دو در \lr{Base} قرار دارند. وضعیت گیره نگهدارنده پهپاد در ابتدا خالی (\lr{Empty}) است. هدف نهایی این است که محموله در موقعیت \lr{Camp} قرار گیرد. اعمالی که پهپاد قادر به انجام آن‌ها است، همراه با پیش‌شرط‌ها و اثرات آن‌ها در زبان \lr{STRIPS} به شرح زیر است:
	
	\vspace{1em}
	\begin{latin}
		\textbf{Fly(x, y)}\\
		Preconditions: At(Drone, x)\\
		Effects: At(Drone, y) $\land$ $\neg$At(Drone, x)\\[1em]
		
		\textbf{Load(c, loc)}\\
		Preconditions: At(Drone, loc) $\land$ At(c, loc) $\land$ Empty(Drone)\\
		Effects: Holding(c, Drone) $\land$ $\neg$At(c, loc) $\land$ $\neg$Empty(Drone)\\[1em]
		
		\textbf{Drop(c, loc)}\\
		Preconditions: At(Drone, loc) $\land$ Holding(c, Drone)\\
		Effects: At(c, loc) $\land$ Empty(Drone) $\land$ $\neg$Holding(c, Drone)
	\end{latin}
	\vspace{1em}
	
	با توجه به این توصیف، به سوالات زیر پاسخ دهید:
	
	\begin{itemize}
		\item[الف)] گراف برنامه‌ریزی (\lr{Planning Graph}) را برای این مسئله رسم کنید.
		\item[ب)] هر ۳ موردی که منجر به \lr{Mutex Exclusion} در دو عمل و هر ۲ موردی که منجر به این شرایط بین دو \lr{literal} می‌شوند را به اختصار توضیح دهید و برای هر پنج مورد یک نمونه \lr{Mutex Link} پیدا کنید.
	\end{itemize}

	\section{\textcolor{AccentGold}{سؤال چهارم}}
	
	رویکرد \lr{SATPlan} مسئله برنامه‌ریزی را به یک مسئله ارضای‌پذیری منطق بولین (\lr{Boolean SAT}) تبدیل می‌کند که توسط حل‌کننده‌های قدرتمند \lr{SAT Solver} قابل حل است. فرض کنید می‌خواهیم یک مسئله برنامه‌ریزی با افق زمانی تعیین‌شده را به این روش فرموله کنیم. به موارد زیر پاسخ دهید:
	
	\begin{itemize}
		\item[الف)] متغیرهای گزاره‌ای (\lr{Propositional Variables}) مورد نیاز برای نمایش وضعیت‌ها (\lr{Fluents}) و اعمال (\lr{Actions}) در هر گام زمانی را به طور دقیق تعریف کنید.
		\item[ب)] برای فرموله‌سازی درست این مسئله، بندهای منطقی (\lr{Clauses}) مربوط به هر یک از موارد زیر را چگونه می‌نویسید؟ (نحوه نگاشت را با زبان ریاضی و منطق گزاره‌ای نشان دهید):
		\begin{itemize}
			\item پیش‌شرط‌ها و اثرات اعمال (\lr{Preconditions \& Effects})
			\item اصول موضوعه چارچوب (\lr{Frame Axioms}) برای حل مسئله قاب (\lr{Frame Problem})
			\item محدودیت‌های طرد متقابل اعمال (\lr{Action Mutex Constraints})
		\end{itemize}
		\item[ج)] به صورت تحلیلی توضیح دهید که چرا با وجود تولید فرمول‌های منطقی بسیار بزرگ و پیچیده، استفاده از حل‌کننده‌های مدرن \lr{SAT} می‌تواند در برخی از دامنه‌های برنامه‌ریزی عملکرد و سرعت بسیار بهتری نسبت به رویکردهای کلاسیک مانند \lr{Graphplan} داشته باشد؟
	\end{itemize}

	\section{\textcolor{AccentGold}{سؤال پنجم}}
	
	در یک خط مونتاژ، یک ربات ثابت با دو بازوی مجزا (بازوی چپ \lr{LeftArm} و بازوی راست \lr{RightArm}) قرار دارد. روی میز کار ربات، دو قطعه وجود دارد: یک لنز (\lr{Lens}) و یک سنسور (\lr{Sensor}). ربات برای ساخت یک دوربین (\lr{Camera})، باید هر دو قطعه را همزمان در دست داشته باشد و سپس آن‌ها را به یکدیگر مونتاژ کند. (در این مسئله نیازی به جابجایی بین مکان‌های مختلف نیست).
	
	\vspace{1em}
	\textbf{هدف ربات:}
	\begin{itemize}
		\item ساخت و مونتاژ نهایی دوربین (\lr{Camera}).
	\end{itemize}
	
	\vspace{1em}
	\textbf{موارد مورد انتظار:}
	\begin{enumerate}
		\item مجموعۀ گزاره‌ها (\lr{Predicates}) موردنیاز مسئله را خودتان به صورت دقیق تعریف کنید.
		\item حالت اولیه (\lr{Initial State}) را مشخص کنید. (دقت کنید که در ابتدا هر دو بازو آزاد هستند و قطعات روی میز قرار دارند).
		\item حالت هدف (\lr{Goal State}) را به صورت مجموعه‌ای از \lr{Literal}ها بنویسید.
		\item اعمالی مانند \lr{PickUp} (برای برداشتن یک قطعه مشخص با یک بازوی مشخص) و \lr{Assemble} (برای مونتاژ کردن دو قطعه‌ای که ربات در دست دارد) را همراه با پیش‌شرط‌ها و اثرات (به سبک \lr{Add/Delete}) تعریف کنید.
		\item سه سطح اول گراف برنامه‌ریزی را رسم کنید:
		\begin{equation*}
			S_0 \rightarrow A_0 \rightarrow S_1
		\end{equation*}
		\item تمام رابطه‌های \lr{Mutex} موجود در سطح‌های $A_0$ و $S_1$ را به دقت مشخص کنید (به ویژه تضادهای موجود در انتخاب بازوها برای قطعات مختلف).
		\item توضیح دهید آیا رسیدن به هدف در سطح $S_1$ امکان‌پذیر است یا نیاز به گسترش سطح‌های بعدی (مثلا $A_1$ و $S_2$) وجود دارد؟ چرا؟
	\end{enumerate}
	
	\vspace{1em}
	\textbf{پیشنهاد (اختیاری):}
	\begin{itemize}
		\item می‌توانید گزاره‌هایی مانند:
		\lr{OnTable(x)}, \lr{Free(arm)}, \lr{Holding(x, arm)}, \lr{Assembled(Camera)} 
		را تعریف کنید.
		\item عمل \lr{PickUp(x, arm)} نیازمند آزاد بودن بازو و روی میز بودن قطعه است.
		\item عمل \lr{Assemble(x, arm1, y, arm2)} نیازمند این است که ربات قطعه اول را در بازوی اول و قطعه دوم را در بازوی دوم نگه داشته باشد. پس از این عمل، بازوها مجدداً آزاد می‌شوند.
		\item در این مسئله به دلیل وجود منابع موازی (دو بازو)، بررسی \lr{Mutex}ها در سطح $A_0$ بسیار مهم است (مثلاً یک بازو نمی‌تواند همزمان دو قطعه را بردارد).
	\end{itemize}
